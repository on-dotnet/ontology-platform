def wait_ready(directory,process):
    deadline=time.monotonic()+20
    while time.monotonic()<deadline:
        rt.require(process.poll() is None,'DIAGNOSTIC_DAEMON_FAILED')
        if (directory/'ingress-modules-ready.json').exists():return
        time.sleep(.1)
    raise ValueError('DIAGNOSTIC_READY_TIMEOUT')
