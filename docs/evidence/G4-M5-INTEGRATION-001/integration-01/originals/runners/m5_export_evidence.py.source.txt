"""Batch the existing V3 byte scanner, then use publish/restore on precisely those approved bytes."""
import sys,json,base64,datetime,struct
from pathlib import Path
sys.path.insert(0,'scripts/maintenance')
import workflow_evidence as evidence
from workflow_execution import run,sha,encoded
root=Path.cwd();task='G4-M5-INTEGRATION-001';rid='integration-01';source=root/'artifacts/workflow'/task;output=root/'docs/evidence'/task/rid;output.mkdir(parents=True,exist_ok=True)
indexpath=output/'originals-index.json'
assert not indexpath.exists(),'Preserve existing export; additions require a new explicitly indexed export.'
files=[]
files.extend(p for p in (source/'commands').glob('*/*/*') if p.is_file() and p.name in ['receipt.json','stdout.txt','stderr.txt'])
files.extend(p for p in (source/'failures').glob('*.json') if p.is_file())
files.extend(source/name for name in ['command-events.jsonl','timeline.jsonl','failures.json'] if (source/name).is_file())
files.extend(p for p in (source/'environment').glob('*/ready.json') if p.is_file())
files.extend(p for p in (root/'artifacts/workflow/policy-qualifications').glob('*.json') if '/'+task+'/' in json.loads(p.read_text()).get('command',{}).get('path',''))
files.extend(p for p in (source/'runs').rglob('*') if p.is_file() and p.suffix in ['.json','.log','.txt'])
runroot=source/rid
files.extend(p for p in runroot.iterdir() if p.is_file() and p.suffix in ['.json','.jsonl','.txt','.png'] and 'compose' not in p.name)
for directory in runroot.iterdir():
 if not directory.is_dir():continue
 if directory.name=='targeted-mutations':files.extend(p for p in directory.rglob('*') if p.is_file() and p.suffix in ['.json','.cs','.targets'])
 elif directory.name.startswith(('web-unit-','web-typecheck-','web-lint-')):files.extend(p for p in directory.iterdir() if p.is_file() and p.suffix in ['.json','.mjs'])
 elif directory.name=='runtime-helper':files.extend(p for p in directory.rglob('*') if p.is_file() and 'obj' not in p.parts and (p.suffix in ['.cs','.csproj'] or p.name.endswith(('.runtimeconfig.json','.deps.json'))))
 elif directory.name=='runtime-snapshots':files.extend(p for p in directory.glob('*.json') if p.is_file())
# Local operational runners are evidence text, not new executable dependencies of the repository.
runners=sorted(Path('/tmp').glob('m5_*.py'))
rows=[]
for path in sorted(set(files+runners)):
 assert path.is_file() and not path.is_symlink()
 raw=path.read_bytes();relative='runners/'+path.name+'.source.txt' if path in runners else (str(path.relative_to(source)) if path.is_relative_to(source) else 'policy-qualifications/'+path.name)+('.source.txt' if path.suffix=='.mjs' else '')
 destination=str((output/'originals'/relative).relative_to(root));rows.append((path,destination,raw))
# Prepare both exactly scanned original bytes and publish's deterministic lossless envelope.
entries=[];approved=set()
for path,destination,raw in rows:
 entries.append(dict(path=destination,bytes=base64.b64encode(raw).decode()))
 if path.suffix=='.png':continue
 try:
  text=raw.decode('utf-8');plain=bool(raw) and '\r' not in text and not text.endswith('\n\n') and '\x1b' not in text and '\x00' not in text and text.endswith('\n') and not any(line.rstrip()!=line for line in text.splitlines())
 except UnicodeDecodeError:plain=False
 if not plain:
  record=dict(originalPath=destination,sha256=sha(raw),sizeBytes=len(raw));payload=json.dumps(dict(encoding='base64',**record,base64=base64.b64encode(raw).decode()),indent=2).encode()+b'\n'
  envelope=str(Path(destination).with_suffix(Path(destination).suffix+'.raw.json'));entries.append(dict(path=envelope,bytes=base64.b64encode(payload).decode()))
for entry in entries:approved.add((entry['path'],sha(base64.b64decode(entry['bytes']))))
manifest=dict(taskId=task,runId=rid,capturedAtUtc=datetime.datetime.now(datetime.timezone.utc).isoformat(),entries=[dict(path=p,sha256=h) for p,h in sorted(approved)])
manifestpath=runroot/'archive-byte-manifest.json';assert not manifestpath.exists();manifestpath.write_text(json.dumps(manifest,indent=2)+'\n')
script=". ./scripts/create-review-package.ps1 -Library; try { $r=[Console]::In.ReadToEnd()|ConvertFrom-Json; foreach($e in $r.entries) { Assert-ReviewPath $e.path; Assert-ReviewContent ([Convert]::FromBase64String($e.bytes)) $e.path }; @{status='PASS';scanned=$r.entries.Count}|ConvertTo-Json -Compress } catch { [Console]::Error.WriteLine($_.Exception.Message); exit 1 }"
result=run(['pwsh','-NoProfile','-Command',script],root,task=task,run_id=rid,stage='m5-archive-original-and-envelope-byte-scan',stdin=json.dumps(dict(entries=entries),separators=(',',':')).encode(),input_key=sha(encoded(manifest)),timeout=120)
print(json.dumps(dict(scanCommandId=result.receipt['commandId'],exitCode=result.returncode,files=len(rows),byteEntries=len(entries))),flush=True)
if result.returncode:print(result.stderr.decode()[-1500:]);sys.exit(1)
assert json.loads(result.stdout)['scanned']==len(entries)
original_preflight=evidence.preflight

 assert repository==root and (relative,sha(raw)) in approved,'Unscanned bytes refused'

try:
 for path,destination,raw in rows:
  if path.suffix=='.png':
   target=root/destination;target.parent.mkdir(parents=True,exist_ok=True)
   with target.open('xb') as stream:stream.write(raw)
   record=dict(originalPath=destination,path=destination,sha256=sha(raw),sizeBytes=len(raw),encoding='raw')
  else:record=evidence.publish(root,destination,raw)
  assert evidence.restore(root,record)==raw
  record['sourcePath']=str(path.relative_to(root)) if path.is_relative_to(root) else str(path)

# Preserve the actual batch scanner receipt separately, with normal per-file scans.

 relative=str((output/'archive-scan'/path.name).relative_to(root));record=evidence.publish(root,relative,path.read_bytes());record['sourcePath']=str(path.relative_to(root));index.append(record)

# Only two representative stable viewports of /analysis; remaining PNGs stay indexed in Git.
shots=[]
for suffix in ['desktop','mobile']:
 path=output/'originals'/rid/('browser-ui-s1-1794833447231-'+suffix+'.png');raw=path.read_bytes();assert raw[:8]==b'\x89PNG\r\n\x1a\n';w,h=struct.unpack('>II',raw[16:24]);shots.append(dict(path=str(path.relative_to(root)),page='/analysis',theme='light',viewport=f'{w}x{h}',stable=True))
selection=output.parent/'review-screenshots.json';assert not selection.exists();selection.write_text(json.dumps(dict(schemaVersion=1,screenshots=shots),indent=2)+'\n')
print(json.dumps(dict(status='EXPORTED_AND_RESTORED',records=len(index),originalBytes=sum(row['sizeBytes'] for row in index),index=str(indexpath.relative_to(root)))),flush=True)
